## Config and run

```yaml title="virig_4imus_3cams_cam_imu.yaml" {1-6,14-19,31-34,40,46,48}
gravity: 9.81
site:
  latitude_deg: 39.68
  longitude_deg: -75.75
  altitude_m: 40
  name: Newark, DE (Univ. of Delaware)
input:
  format: ros1
  path: virig_4imus_3cams_april_01.bag
  cam_topics:
  - /blackfly/image_raw
  - /t265/fisheye1/image_raw
  - /t265/fisheye2/image_raw
  imu_topics:
  - /imu/data
  - /gx3_25/data
  - /gx3_35/imu/data
  - /t265/imu
  window_mode: motion
  segment_length: 80.0
board:
  board_type: apriltag
  rows: 6
  cols: 6
  tag_size: 0.088
  tag_spacing: 0.3
  tag_family: t36h11
  heights_file: null
camera_input:
  path: mvis_4imus_3cams_calib.yaml
  cameras:
  - {}
  - factor_hz: 4.0
  - factor_hz: 4.0
solve:
  max_iterations: 40
  corner_subsample: two_anti
  estimate_t_ga: true
  sigma_t_ga: 0.01
  sigma_g: 0.00065
  sigma_a: 0.0026
  sigma_bg: 0.001
  sigma_ba: 0.002
  vision_pose: imu
estimate:
  board_points: true
  camera:
    intrinsics: true
    tr: false
    td: true
    extrinsics: true
  imu:
    intrinsics: dw_da
    model: calyx
    extrinsics: true
    td: true
output:
  path: cam_imu.json
```

```bash title="shell"
mvis cam-imu -c virig_4imus_3cams_cam_imu.yaml
```
